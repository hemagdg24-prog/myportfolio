import 'package:flutter/material.dart';
import 'package:heal90/services/api.dart';

// 90 Heal Healthcare App Module by Hemaaswini
class HealthDashboard extends StatelessWidget {
  const HealthDashboard({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('90 Heal Portal')),
      body: FutureBuilder<List<Patient>>(
        future: ApiService.fetchPatients(),
        builder: (context, snapshot) {
          if (!snapshot.hasData) {
            return const Center(child: CircularProgressIndicator());
          }
          return ListView.builder(
            itemCount: snapshot.data!.length,
            itemBuilder: (ctx, i) => PatientCard(record: snapshot.data![i]),
          );
        },
      ),
    );
  }
}
